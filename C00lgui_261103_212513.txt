-- Dragging Mobile Support
local dragging, dragStart, startPos
TopBar.InputBegan:Connect(function(input)
if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
dragging = true
dragStart = input.Position
startPos = Main.Position
end
end)
UserInputService.InputChanged:Connect(function(input)
if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
local delta = input.Position - dragStart
Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end
end)
UserInputService.InputEnded:Connect(function(input)
if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
dragging = false
end
end)

local root = char.HumanoidRootPart
if flying then

bg.Name = "C00lG"
bg.MaxTorque = Vector3.new(9e9,9e9,9e9)
bg.CFrame = root.CFrame
local bv = Instance.new("BodyVelocity", root)
bv.Name = "C00lV"
bv.MaxForce = Vector3.new(9e9,9e9,9e9)
bv.Velocity = Vector3.new(0,0,0)
task.spawn(function()
while flying and char and root do
local cam = workspace.CurrentCamera
local move = char.Humanoid.MoveDirection
if move.Magnitude > 0 then
bv.Velocity = cam.CFrame.LookVector * 50 + Vector3.new(0,2,0)
else
bv.Velocity = Vector3.new(0,0.1,0)
end
bg.CFrame = cam.CFrame
task.wait()
end
if bg then bg:Destroy() end
if bv then bv:Destroy() end
end)
else
if root:FindFirstChild("C00lG") then root.C00lG:Destroy() end
if root:FindFirstChild("C00lV") then root.C00lV:Destroy() end
end
end)
