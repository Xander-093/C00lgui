-- Fungsi Dragging Jendela Utama
local dragging, dragStart, startPos
TopBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        if dragging then
            local delta = input.Position - dragStart
            MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end
end)

createButton("🚀 Toggle Fly (Terbang)", function()
    flying = not flying
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return end
    local rootPart = char.HumanoidRootPart
    
    if flying then
        local bg = Instance.new("BodyGyro", rootPart)
        bg.Name = "FlyGyro"
        bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
        bg.CFrame = rootPart.CFrame
        
        local bv = Instance.new("BodyVelocity", rootPart)
        bv.Name = "FlyVelocity"
        bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
        bv.Velocity = Vector3.new(0, 0, 0)
        
        task.spawn(function()
            while flying and char and rootPart.Parent do
                local cam = workspace.CurrentCamera
                local vel = Vector3.new()
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then vel = vel + cam.CoordinateFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then vel = vel - cam.CoordinateFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then vel = vel - cam.CoordinateFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then vel = vel + cam.CoordinateFrame.RightVector end
                bv.Velocity = vel * flySpeed
                bg.CFrame = cam.CoordinateFrame
                task.wait()
            end
            if rootPart:FindFirstChild("FlyGyro") then rootPart.FlyGyro:Destroy() end
            if rootPart:FindFirstChild("FlyVelocity") then rootPart.FlyVelocity:Destroy() end
        end)
    end
end)

-- Eksekusi Fling & Auto-Return
FlingBox.FocusLost:Connect(function(enterPressed)
    if not enterPressed then return end
    local query = string.lower(FlingBox.Text)
    local targetPlayer = nil
    
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and (string.sub(string.lower(p.Name), 1, #query) == query or string.sub(string.lower(p.DisplayName), 1, #query) == query) then
            targetPlayer = p
            break
        end
    end
    
    if targetPlayer and targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") then
        local char = LocalPlayer.Character
        if not char or not char:FindFirstChild("HumanoidRootPart") then return end
        local rootPart = char.HumanoidRootPart
        local originalPos = rootPart.CFrame -- Simpan posisi awal untuk kembali
        
        -- 1. Nyalakan Noclip Otomatis
        local noclipConn
        noclipConn = RunService.Stepped:Connect(function()
            if char then
                for _, part in pairs(char:GetDescendants()) do
                    if part:IsA("BasePart") then part.CanCollide = false end
                end
            end
        end)
        
        -- 2. Teleport ke Target
        local targetRoot = targetPlayer.Character.HumanoidRootPart
        rootPart.CFrame = targetRoot.CFrame
        task.wait(0.1)
        
        -- 3. Fling Process (Putar-putar kecepatan tinggi)
        local bav = Instance.new("BodyAngularVelocity", rootPart)
        bav.MaxTorque = Vector3.new(0, math.huge, 0)
        bav.AngularVelocity = Vector3.new(0, 99999, 0)
        
        local bavPos = Instance.new("BodyVelocity", rootPart)
        bavPos.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        bavPos.Velocity = Vector3.new(0, 50, 0)
        
        task.wait(0.6) -- Durasi Fling
        
        if bav then bav:Destroy() end
        if bavPos then bavPos:Destroy() end
        
        -- 4. Matikan Noclip & Kembali ke Spot Awal
        if noclipConn then noclipConn:Disconnect() end
        rootPart.CFrame = originalPos
        FlingBox.Text = "Berhasil Fling!"
        task.wait(2)
        FlingBox.Text = ""
    else
        FlingBox.Text = "Player tidak ketemu!"
        task.wait(2)
        FlingBox.Text = ""
    end
end)
